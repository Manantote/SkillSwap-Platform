1. Database Schema (MySQL)

Core relational database tables defining users, skills, and the learning session progress tracking system.

config/schema.sql (Excerpt)

CREATE TABLE IF NOT EXISTS users (
    id           INT          NOT NULL AUTO_INCREMENT,
    firebase_uid VARCHAR(128) NOT NULL UNIQUE,
    name         VARCHAR(100) NOT NULL,
    email        VARCHAR(200) NOT NULL,
    created_at   TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
    is_online    TINYINT(1)   DEFAULT 0,
    PRIMARY KEY (id)
);

CREATE TABLE IF NOT EXISTS learning_progress (
    id                  INT  NOT NULL AUTO_INCREMENT,
    user_id             INT  NOT NULL,
    skill_id            INT  NOT NULL,
    sessions_completed  INT  NOT NULL DEFAULT 0,
    total_time          INT  NOT NULL DEFAULT 0,   -- minutes
    status              ENUM('learning','completed') NOT NULL DEFAULT 'learning',
    last_updated        TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_user_skill (user_id, skill_id),
    FOREIGN KEY (user_id)  REFERENCES users(id) ON DELETE CASCADE
);

2. Backend Core (PHP)

Database connection handler and secure Firebase UID validation middleware.

config/db.php (Excerpt)

<?php
define('DB_HOST', 'localhost');
define('DB_NAME', 'skillswap_new');
define('DB_USER', 'root');
define('DB_PASS', '');

function getDB(): PDO {
    static $pdo = null;
    if ($pdo === null) {
        $dsn = 'mysql:host='.DB_HOST.';dbname='.DB_NAME.';charset=utf8mb4';
        $options = [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        ];
        $pdo = new PDO($dsn, DB_USER, DB_PASS, $options);
    }
    return $pdo;
}

// Security Middleware: Requires Firebase UID token from request headers
function requireUID(): string {
    $uid = $_SERVER['HTTP_X_FIREBASE_UID'] ?? null;
    if (!$uid) {
        http_response_code(401);
        echo json_encode(['success' => false, 'error' => 'Authentication required.']);
        exit;
    }
    return trim($uid);
}
?>

3. Business Logic API (PHP)

API endpoint handling the completion of a video session, calculating duration, and auto-updating learning progress.

api/end_session.php (Excerpt)

<?php
require_once __DIR__ . '/../config/db.php';
$learnerUID = requireUID();
$pdo = getDB();

$body = json_decode(file_get_contents('php://input'), true);
$sessionId = (int)$body['session_id'];

// Calculate Duration
$now      = new DateTime();
$start    = new DateTime($session['start_time']);
$duration = (int)round(($now->getTimestamp() - $start->getTimestamp()) / 60);

// Validate minimum time requirement
$status = $duration >= 15 ? 'completed' : 'invalid';

$upd = $pdo->prepare('UPDATE sessions SET end_time = ?, duration_minutes = ?, status = ? WHERE id = ?');
$upd->execute([$now->format('Y-m-d H:i:s'), $duration, $status, $sessionId]);

// Auto-update user's learning progress if valid
if ($status === 'completed') {
    $stmt = $pdo->prepare('
        INSERT INTO learning_progress (user_id, skill_id, sessions_completed, total_time, status)
        VALUES (?, ?, 1, ?, "learning")
        ON DUPLICATE KEY UPDATE
           sessions_completed = sessions_completed + 1,
           total_time         = total_time + VALUES(total_time)
    ');
    $stmt->execute([$learnerId, $skillId, $duration]);
}
?>

4. Client-Side Logic (JavaScript)

Frontend asynchronous fetch function interacting with the backend and managing UI state for quizzes.

js/session-quiz.js (Excerpt)

// Fetch wrapper attaching secure Firebase Auth headers
async function authFetch(url, options = {}) {
    const uid = await getUID(); // Wait for Firebase Auth State
    options.headers = { 
        ...(options.headers || {}), 
        'X-Firebase-UID': uid, 
        'Content-Type': 'application/json' 
    };
    const res = await fetch(url, options);
    return res.json();
}

// Submits the completed quiz to the backend for secure grading
window.submitQuizAnswers = async function() {
    const btn = document.getElementById('quiz-submit-btn');
    btn.innerHTML = 'Grading...';

    const data = await authFetch('../api/submit_quiz.php', {
        method: 'POST',
        body: JSON.stringify({
            attempt_id: quizState.attemptId,
            quiz_token: quizState.token,
            answers: quizState.answers,
        }),
    });

    if (data.success && data.data.passed) {
        document.getElementById('quiz-result-title').textContent = 'Congratulations!';
        // Reveal dynamically generated printable certificate link
        const certLink = `../api/generate_certificate.php?token=${data.data.certificate.cert_token}`;
        document.getElementById('quiz-cert-link').href = certLink;
    }
};
